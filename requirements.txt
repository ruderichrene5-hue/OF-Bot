# --- numpy / OpenCV -----------------------------------------------------------
# Do NOT re-pin these to exact versions. numpy==1.26.4 and opencv-python==4.10.0.84
# (the old pins) have no prebuilt wheels for Python 3.13, so pip falls back to a
# from-source build that needs a C/C++ toolchain -- it fails on a clean Windows
# box and, on Linux, either fails or silently spends ~20 minutes compiling.
# packaging/requirements-windows.txt existed purely to work around that; these
# ranges fix it at the source. numpy is held at 2.x because OpenCV 5 is built
# against the numpy 2 ABI, and mixing the two crashes at import.
numpy>=2.1,<3

# On a headless Linux server the regular opencv-python wheel fails to import: it
# links libGL.so.1, which a server with no desktop does not have. That import is
# wrapped in a try/except (flows/instagram.py), so the failure is SILENT -- cv2
# becomes None and the blue-button classification and OCR verification just stop
# working, with the flows still reporting success. The headless wheel is the same
# library minus the GUI dependencies (verified: it needs only libc/libstdc++/libz).
opencv-python-headless>=4.10,<6; sys_platform == "linux"
opencv-python>=4.10,<6; sys_platform != "linux"

pytesseract==0.3.13
Pillow>=10.0.0

# Pulls in adbutils, lxml, retry2 -- all prebuilt wheels, no system libs needed.
